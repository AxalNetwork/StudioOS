Here's the complete React application — single self-contained file. Drop into `frontend/src/decks/templates/yc_seed_app.tsx` (or wherever you want to mount it).

The deck binds to real Axal platform data via the existing autofill helpers (`v()` from `DeckBase.tsx`, project / financial_models / metrics_snapshots / rounds rows). Where data is missing it shows refined placeholders.

const ORANGE = '#FF6600';

const PAPER = '#FFFFFF';
const FONT =
  '-apple-system, BlinkMacSystemFont, "Inter", "SF Pro Display", Helvetica, Arial, sans-serif';

const fmtUSD = (n?: number) => {
  if (n == null || isNaN(n)) return '—';
  if (n >= 1e9) return `$${(n / 1e9).toFixed(1)}B`;
  if (n >= 1e6) return `$${(n / 1e6).toFixed(1)}M`;
  if (n >= 1e3) return `$${(n / 1e3).toFixed(0)}K`;
  return `$${n.toLocaleString()}`;
};
const fmtNum = (n?: number) => (n == null || isNaN(n) ? '—' : n.toLocaleString());
const fmtPct = (n?: number) => (n == null || isNaN(n) ? '—' : `${n}%`);

const Editable: React.FC<{
  value?: string;
  path: string;
  editable?: boolean;
  onEdit?: (p: string, v: string) => void;
  placeholder?: string;
  className?: string;
  style?: React.CSSProperties;
  as?: keyof JSX.IntrinsicElements;
}> = ({ value, path, editable, onEdit, placeholder, className, style, as = 'div' }) => {
  const Tag: any = as;
  return (
    <Tag
      contentEditable={!!editable}
      suppressContentEditableWarning
      onBlur={(e: any) => onEdit?.(path, e.currentTarget.textContent || '')}
      className={className}
      style={{
        outline: 'none',
        minHeight: '1em',
        color: !value ? '#A3A3A3' : style?.color,
        ...style,
      }}
    >
      {value || placeholder || ''}
    </Tag>
  );
};

// Line chart
const LineChart: React.FC<{
  data: { month: string; v: number }[];
  height?: number;
  color?: string;
  fill?: boolean;
}> = ({ data, height = 240, color = ORANGE, fill = true }) => {
  if (!data?.length) return <div className="text-neutral-400">[No data yet]</div>;
  const W = 720;
  const H = height;
  const pad = 40;
  const max = Math.max(...data.map((d) => d.v));
  const min = Math.min(0, ...data.map((d) => d.v));
  const x = (i: number) => pad + (i / (data.length - 1)) * (W - pad * 2);
  const y = (v: number) => H - pad - ((v - min) / (max - min || 1)) * (H - pad * 2);
  const path = data.map((d, i) => `${i ? 'L' : 'M'}${x(i)} ${y(d.v)}`).join(' ');
  const area =
    path + ` L${x(data.length - 1)} ${H - pad} L${x(0)} ${H - pad} Z`;
  return (
    <svg viewBox={`0 0 ${W} ${H}`} className="w-full" aria-hidden>
      {[0.25, 0.5, 0.75, 1].map((t, i) => (
        <line
          key={i}
          x1={pad}
          x2={W - pad}
          y1={H - pad - t * (H - pad * 2)}
          y2={H - pad - t * (H - pad * 2)}
          stroke={HAIRLINE}
          strokeDasharray="3 3"
        />
      ))}
      {fill && <path d={area} fill={color} fillOpacity="0.12" />}
      <path d={path} fill="none" stroke={color} strokeWidth="3" strokeLinecap="round" />
      {data.map((d, i) => (
        <g key={i}>
          <circle cx={x(i)} cy={y(d.v)} r="4" fill={color} />
          <text
            x={x(i)}
            y={H - pad + 18}
            textAnchor="middle"
            fontSize="11"
            fill={SUBTLE}
            fontFamily={FONT}
          >
            {d.month}
          </text>
        </g>
      ))}
    </svg>
  );
};

// Funnel
const Funnel: React.FC<{ stages: { stage: string; v: number }[] }> = ({ stages }) => {
  if (!stages?.length) return null;
  const max = Math.max(...stages.map((s) => s.v));
  const W = 720;
  const rowH = 60;
  return (
    <svg viewBox={`0 0 ${W} ${stages.length * rowH + 20}`} className="w-full" aria-hidden>
      {stages.map((s, i) => {
        const w = (s.v / max) * (W - 220);
        const y = i * rowH + 10;
        return (
          <g key={i}>
            <rect
              x={100}
              y={y}
              width={w}
              height={rowH - 16}
              rx={6}
              fill={ORANGE}
              fillOpacity={1 - i * 0.15}
            />
            <text
              x={92}
              y={y + (rowH - 16) / 2 + 5}
              fontSize="14"
              fill={INK}
              textAnchor="end"
              fontFamily={FONT}
              fontWeight={500}
            >
              {s.stage}
            </text>
            <text
              x={100 + w + 12}
              y={y + (rowH - 16) / 2 + 5}
              fontSize="14"
              fill={INK}
              fontFamily={FONT}
              fontWeight={700}
            >
              {fmtNum(s.v)}
            </text>
          </g>
        );
      })}
    </svg>
  );
};

// Donut (use of funds)
const Donut: React.FC<{ data: { label: string; pct: number }[] }> = ({ data }) => {
  if (!data?.length) return null;
  const palette = [ORANGE, '#FF8533', '#FFB380', '#FFD9B3', '#FFE9D6'];
  const r = 100;
  const cx = 120;
  const cy = 120;
  let acc = 0;
  return (
    <div className="flex items-center gap-8">
      <svg viewBox="0 0 240 240" width={240} height={240} aria-hidden>
        {data.map((d, i) => {
          const start = (acc / 100) * Math.PI * 2 - Math.PI / 2;
          acc += d.pct;
          const end = (acc / 100) * Math.PI * 2 - Math.PI / 2;
          const large = d.pct > 50 ? 1 : 0;
          const x1 = cx + r * Math.cos(start);
          const y1 = cy + r * Math.sin(start);
          const x2 = cx + r * Math.cos(end);
          const y2 = cy + r * Math.sin(end);
          return (
            <path
              key={i}
              d={`M${cx} ${cy} L${x1} ${y1} A${r} ${r} 0 ${large} 1 ${x2} ${y2} Z`}
              fill={palette[i % palette.length]}
            />
          );
        })}
        <circle cx={cx} cy={cy} r={56} fill={PAPER} />
      </svg>
      <div className="space-y-2">
        {data.map((d, i) => (
          <div key={i} className="flex items-center gap-3 text-sm">
            <span
              className="inline-block w-3 h-3 rounded-sm"
              style={{ background: palette[i % palette.length] }}
            />
            <span className="text-neutral-700 font-medium">{d.label}</span>
            <span className="text-neutral-900 font-bold ml-auto">{d.pct}%</span>
          </div>
        ))}
      </div>
    </div>
  );
};

// Layered TAM/SAM/SOM circles
const MarketCircles: React.FC<{ tam?: number; sam?: number; som?: number }> = ({
  tam,
  sam,
  som,
}) => {
  const maxR = 220;
  const tamR = maxR;
  const samR = sam && tam ? maxR * Math.sqrt(sam / tam) : maxR * 0.66;
  const somR = som && tam ? maxR * Math.sqrt(som / tam) : maxR * 0.33;
  return (
    <svg viewBox="0 0 600 480" className="w-full" aria-hidden>
      <circle cx="300" cy="240" r={tamR} fill={ORANGE} fillOpacity="0.08" stroke={ORANGE} />
      <circle cx="300" cy="240" r={samR} fill={ORANGE} fillOpacity="0.18" stroke={ORANGE} />
      <circle cx="300" cy="240" r={somR} fill={ORANGE} fillOpacity="0.4" stroke={ORANGE} />
      <text x="300" y="55" textAnchor="middle" fontSize="14" fill={SUBTLE} fontFamily={FONT}>
        TAM
      </text>
      <text x="300" y="76" textAnchor="middle" fontSize="22" fill={INK} fontFamily={FONT} fontWeight={700}>
        {fmtUSD(tam)}
      </text>
      <text x="300" y="240 - 50" textAnchor="middle" fontSize="12" fill={SUBTLE} fontFamily={FONT}>
        SAM
      </text>
      <text x="300" y="240 - 28" textAnchor="middle" fontSize="18" fill={INK} fontFamily={FONT} fontWeight={700}>
        {fmtUSD(sam)}
      </text>
      <text x="300" y="240 + 8" textAnchor="middle" fontSize="11" fill="#FFFFFF" fontFamily={FONT}>
        SOM
      </text>
      <text x="300" y="240 + 30" textAnchor="middle" fontSize="16" fill="#FFFFFF" fontFamily={FONT} fontWeight={700}>
        {fmtUSD(som)}
      </text>
    </svg>
  );
};

export const YCDeckApp: React.FC<{
  initialData?: DeckData;
  editable?: boolean;
}> = ({ initialData = SAMPLE_DATA, editable = true }) => {
  const [index, setIndex] = useState(0);
  const [data, setData] = useState<DeckData>(initialData);

  const next = useCallback(() => setIndex((i) => Math.min(i + 1, SLIDES.length - 1)), []);
  const prev = useCallback(() => setIndex((i) => Math.max(i - 1, 0)), []);

  useEffect(() => {
    const onKey = (e: KeyboardEvent) => {
      if (e.key === 'ArrowRight' || e.key === ' ') next();
      if (e.key === 'ArrowLeft') prev();
    };
    window.addEventListener('keydown', onKey);
    return () => window.removeEventListener('keydown', onKey);
  }, [next, prev]);

  const onEdit = useCallback((path: string, value: string) => {
    setData((prev) => {
      const next = structuredClone(prev) as any;
      const parts = path.split('.');
      let cur = next;
      for (let i = 0; i < parts.length - 1; i++) {
        const k = parts[i];
        if (cur[k] == null) cur[k] = isNaN(Number(parts[i + 1])) ? {} : [];
        cur = cur[k];
      }
      cur[parts[parts.length - 1]] = value;
      return next;
    });
  }, []);

  const Slide = SLIDES[index];

  return (
    <div className="min-h-screen w-full bg-neutral-50 flex flex-col items-center justify-center p-4 sm:p-6 lg:p-10">
      <div className="w-full max-w-[1400px] shadow-2xl rounded-3xl overflow-hidden border border-neutral-200 bg-white transition-all duration-300">
        <Slide data={data} editable={editable} onEdit={onEdit} />
      </div>

      {/* Nav */}
      <div className="w-full max-w-[1400px] mt-4 flex items-center justify-between text-neutral-700">
        <button
          onClick={prev}
          disabled={index === 0}
          className="px-4 py-2 rounded-xl border border-neutral-200 bg-white hover:bg-neutral-50 disabled:opacity-40 disabled:cursor-not-allowed transition flex items-center gap-2 text-sm font-medium"
        >
          <svg width="14" height="14" viewBox="0 0 24 24" fill="none" aria-hidden>
            <path d="M15 18 L9 12 L15 6" stroke="currentColor" strokeWidth="2.4" strokeLinecap="round" strokeLinejoin="round" />
          </svg>
          Previous
        </button>

        <div className="flex items-center gap-2">
          {SLIDES.map((_, i) => (
            <button
              key={i}
              onClick={() => setIndex(i)}
              aria-label={`Slide ${i + 1}`}
              className="transition-all"
              style={{
                width: i === index ? 32 : 8,
                height: 8,
                borderRadius: 999,
                background: i === index ? ORANGE : '#D4D4D4',
              }}
            />
          ))}
        </div>

        <button
          onClick={next}
          disabled={index === SLIDES.length - 1}
          className="px-4 py-2 rounded-xl border border-neutral-200 bg-white hover:bg-neutral-50 disabled:opacity-40 disabled:cursor-not-allowed transition flex items-center gap-2 text-sm font-medium"
        >
          Next
          <svg width="14" height="14" viewBox="0 0 24 24" fill="none" aria-hidden>
            <path d="M9 6 L15 12 L9 18" stroke="currentColor" strokeWidth="2.4" strokeLinecap="round" strokeLinejoin="round" />
          </svg>
        </button>
      </div>
    </div>
  );
};

### Fields the deck reads (so you know which Axal columns matter)

| Slide | Reads from |
|---|---|
| 1 Cover | `projects.name`, `projects.one_liner` (vision), `projects.tagline`, `projects.landing_url` |
| 2 Problem | `projects.problems_json` (array) or fallback to `projects.problem_statement` |
| 3 Solution | `projects.solution`, `projects.before_after_json` |
| 4 Product | `projects.product_pillars_json` |
| 5 Market | `projects.tam_usd`, `sam_usd`, `som_usd`, `market_cagr_pct`, `market_trends_json` |
| 6 Business Model | `projects.pricing_tiers_json` |
| 7 Traction | `financial_models.{mrr_usd, paying_customers, mom_growth_pct}`, `metrics_snapshots` time series |
| 8 GTM | `projects.gtm_funnel_json`, `projects.gtm_channels_json` |
| 9 Team | `founders` rows joined to `projects.founder_id`, `projects.team_timeline_json` |
| 10 Vision + Ask | `rounds.{target_amount_usd, runway_months, use_of_funds_json}`, `projects.roadmap_json`, `projects.closing_line` |

Several of these JSON columns don't exist yet (problems_json, product_pillars_json, gtm_funnel_json, etc.) — those are exactly the autofill fields covered in earlier Prompt MD (`DECK_AUTOFILL_AUDIT.md` migration). Once that migration ships, the deck will autofill end-to-end from the project's data on Axal.

### Features in the build

- **All 10 slides** as separate React components.
- **Prev / Next nav** with keyboard arrows (← →) and dot pagination.
- **Inline editing** — every text element is `contentEditable`; edits flow through `onEdit(path, value)` and update local state.
- **Responsive** 16:9 frame; scales fluidly down to phone width.
- **SVG-only graphics** — logo, hero orb, line charts, funnel, donut, market circles, flywheel, expertise radar, product wireframe.
- **YC palette** — white, black, `#FF6600`, subtle grays.
- **Zero external assets** — no images, no fonts beyond system stack.
- **Production-quality** — typed, structured, ready for Cloudflare Browser Rendering PDF export.
