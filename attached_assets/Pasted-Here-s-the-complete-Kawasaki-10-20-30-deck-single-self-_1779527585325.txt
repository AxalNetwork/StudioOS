Here's the complete Kawasaki 10/20/30 deck — single self-contained React + TypeScript + Tailwind file with Framer Motion. Drop into `frontend/src/decks/templates/kawasaki_10_20_30_app.tsx`.

Data binds to the same Axal platform rows used by the YC and Sequoia decks; falls back to richly-populated `SAMPLE_DATA` when rendered standalone.

const fmtUSD = (n?: number) => {
  if (n == null || isNaN(n)) return '—';
  if (n >= 1e9) return `$${(n / 1e9).toFixed(1)}B`;
  if (n >= 1e6) return `$${(n / 1e6).toFixed(1)}M`;
  if (n >= 1e3) return `$${(n / 1e3).toFixed(0)}K`;
  return `$${n.toLocaleString()}`;
};
const fmtNum = (n?: number) => (n == null || isNaN(n) ? '—' : n.toLocaleString());
const fmtPct = (n?: number) => (n == null || isNaN(n) ? '—' : `${n}%`);

const Editable: React.FC<{
  value?: string;
  path: string;
  editable?: boolean;
  onEdit?: (p: string, v: string) => void;
  placeholder?: string;
  className?: string;
  style?: React.CSSProperties;
  as?: keyof JSX.IntrinsicElements;
}> = ({ value, path, editable, onEdit, placeholder, className, style, as = 'div' }) => {
  const Tag: any = as;
  return (
    <Tag
      contentEditable={!!editable}
      suppressContentEditableWarning
      onBlur={(e: any) => onEdit?.(path, e.currentTarget.textContent || '')}
      className={className}
      style={{
        outline: 'none',
        minHeight: '1em',
        color: !value ? '#A3A3A3' : style?.color,
        ...style,
      }}
    >
      {value || placeholder || ''}
    </Tag>
  );
};

const RevenueFlow: React.FC<{ flow: { from: string; to: string; label?: string }[] }> = ({ flow }) => {
  // 3-node linear flow; keep extremely simple per Kawasaki rules
  const nodes = ['Customer', ...flow.map((f) => f.to)];
  const W = 1100;
  const H = 240;
  const cx = (i: number) => 120 + i * ((W - 240) / Math.max(nodes.length - 1, 1));
  return (
    <svg viewBox={`0 0 ${W} ${H}`} className="w-full" aria-hidden>
      {nodes.map((n, i) => (
        <g key={i}>
          <rect
            x={cx(i) - 110}
            y={H / 2 - 50}
            width={220}
            height={100}
            rx={14}
            fill={i === 0 ? PAPER : i === nodes.length - 1 ? ACCENT : PAPER_DEEP}
            stroke={i === 0 ? INK : i === nodes.length - 1 ? ACCENT : HAIRLINE}
            strokeWidth={i === 0 ? 2 : 1}
          />
          <text
            x={cx(i)}
            y={H / 2 + 8}
            textAnchor="middle"
            fontSize="22"
            fontFamily={FONT}
            fontWeight={i === nodes.length - 1 ? 700 : 600}
            fill={i === nodes.length - 1 ? '#FFFFFF' : INK}
          >
            {n}
          </text>
        </g>
      ))}
      {nodes.slice(0, -1).map((_, i) => (
        <g key={'arr' + i}>
          <line
            x1={cx(i) + 112}
            x2={cx(i + 1) - 112}
            y1={H / 2}
            y2={H / 2}
            stroke={ACCENT}
            strokeWidth={3}
          />
          <path
            d={`M${cx(i + 1) - 124} ${H / 2 - 9} L${cx(i + 1) - 112} ${H / 2} L${cx(i + 1) - 124} ${H / 2 + 9}`}
            fill="none"
            stroke={ACCENT}
            strokeWidth={3}
            strokeLinejoin="round"
            strokeLinecap="round"
          />
          {flow[i]?.label && (
            <text
              x={(cx(i) + cx(i + 1)) / 2}
              y={H / 2 - 18}
              textAnchor="middle"
              fontSize="18"
              fontFamily={FONT}
              fontWeight={600}
              fill={ACCENT}
            >
              {flow[i].label}
            </text>
          )}
        </g>
      ))}
    </svg>
  );
};

const RevenueChart: React.FC<{ data: { label: string; v: number }[]; color?: string }> = ({
  data,
  color = ACCENT,
}) => {
  if (!data?.length) return null;
  const W = 1100;
  const H = 340;
  const padL = 64;
  const padR = 40;
  const padT = 40;
  const padB = 60;
  const max = Math.max(...data.map((d) => d.v));
  const x = (i: number) => padL + (i / (data.length - 1)) * (W - padL - padR);
  const y = (v: number) => H - padB - (v / max) * (H - padT - padB);
  const path = data.map((d, i) => `${i ? 'L' : 'M'}${x(i)} ${y(d.v)}`).join(' ');
  const area = path + ` L${x(data.length - 1)} ${H - padB} L${x(0)} ${H - padB} Z`;
  return (
    <svg viewBox={`0 0 ${W} ${H}`} className="w-full" aria-hidden>
      {[0.25, 0.5, 0.75, 1].map((t, i) => (
        <line key={i} x1={padL} x2={W - padR} y1={H - padB - t * (H - padT - padB)} y2={H - padB - t * (H - padT - padB)} stroke={HAIRLINE} strokeDasharray="4 4" />
      ))}
      <path d={area} fill={color} fillOpacity="0.12" />
      <path d={path} fill="none" stroke={color} strokeWidth={4} strokeLinecap="round" strokeLinejoin="round" />
      {data.map((d, i) => (
        <g key={i}>
          <circle cx={x(i)} cy={y(d.v)} r={6} fill={color} />
          <text x={x(i)} y={H - padB + 30} textAnchor="middle" fontSize="20" fontFamily={FONT} fill={SUBTLE}>
            {d.label}
          </text>
        </g>
      ))}
    </svg>
  );
};

const Donut: React.FC<{ data: { label: string; pct: number }[] }> = ({ data }) => {
  if (!data?.length) return null;
  const palette = [ACCENT, '#3D87FF', '#7CA9FF', '#B6CCFF', '#DBE6FF'];
  const r = 130;
  const cx = 150;
  const cy = 150;
  let acc = 0;
  return (
    <div className="flex items-center gap-12 w-full">
      <svg viewBox="0 0 300 300" width={300} height={300} aria-hidden>
        {data.map((d, i) => {
          const start = (acc / 100) * Math.PI * 2 - Math.PI / 2;
          acc += d.pct;
          const end = (acc / 100) * Math.PI * 2 - Math.PI / 2;
          const large = d.pct > 50 ? 1 : 0;
          const x1 = cx + r * Math.cos(start);
          const y1 = cy + r * Math.sin(start);
          const x2 = cx + r * Math.cos(end);
          const y2 = cy + r * Math.sin(end);
          return (
            <path
              key={i}
              d={`M${cx} ${cy} L${x1} ${y1} A${r} ${r} 0 ${large} 1 ${x2} ${y2} Z`}
              fill={palette[i % palette.length]}
            />
          );
        })}
        <circle cx={cx} cy={cy} r={70} fill={PAPER} />
      </svg>
      <div className="space-y-4 flex-1">
        {data.map((d, i) => (
          <div key={i} className="flex items-center gap-4" style={{ fontSize: TYPE.detail }}>
            <span
              className="inline-block w-4 h-4 rounded-sm"
              style={{ background: palette[i % palette.length] }}
            />
            <span style={{ color: INK, fontWeight: 600 }}>{d.label}</span>
            <span style={{ color: INK, fontWeight: 800, marginLeft: 'auto' }}>{d.pct}%</span>
          </div>
        ))}
      </div>
    </div>

const SLIDES: React.FC<DeckProps>[] = [
  Slide1Problem,
  Slide2Solution,
  Slide3BusinessModel,
  Slide4Magic,
  Slide5MarketingSales,
  Slide6Competition,
  Slide7Team,
  Slide8Projections,
  Slide9CurrentStatus,
  Slide10Ask,
];

export const KawasakiDeckApp: React.FC<{
  initialData?: KawasakiData;
  editable?: boolean;
}> = ({ initialData = SAMPLE_DATA, editable = true }) => {
  const [index, setIndex] = useState(0);
  const [data, setData] = useState<KawasakiData>(initialData);
  const reduced = useReducedMotion();

  const next = useCallback(() => setIndex((i) => Math.min(i + 1, SLIDES.length - 1)), []);
  const prev = useCallback(() => setIndex((i) => Math.max(i - 1, 0)), []);

  useEffect(() => {
    const onKey = (e: KeyboardEvent) => {
      if (e.key === 'ArrowRight' || e.key === ' ') { e.preventDefault(); next(); }
      if (e.key === 'ArrowLeft') { e.preventDefault(); prev(); }
    };
    window.addEventListener('keydown', onKey);
    return () => window.removeEventListener('keydown', onKey);
  }, [next, prev]);

  const onEdit = useCallback((path: string, value: string) => {
    setData((prev) => {
      const next = structuredClone(prev) as any;
      const parts = path.split('.');
      let cur = next;
      for (let i = 0; i < parts.length - 1; i++) {
        const k = parts[i];
        const isArrKey = !isNaN(Number(parts[i + 1]));
        if (cur[k] == null) cur[k] = isArrKey ? [] : {};
        cur = cur[k];
      }
      cur[parts[parts.length - 1]] = value;
      return next;
    });
  }, []);

  const Slide = SLIDES[index];

  return (
    <div className="min-h-screen w-full flex flex-col items-center justify-center p-4 sm:p-6 lg:p-10"
      style={{ background: '#FAFAFA', fontFamily: FONT }}>
      <div
        className="w-full max-w-[1400px] rounded-3xl overflow-hidden border bg-white shadow-[0_30px_80px_-30px_rgba(0,0,0,0.25)]"
        style={{ borderColor: HAIRLINE }}
      >
        <AnimatePresence mode="wait">
          <motion.div
            key={index}
            initial={reduced ? false : { opacity: 0, y: 18 }}
            animate={reduced ? {} : { opacity: 1, y: 0 }}
            exit={reduced ? {} : { opacity: 0, y: -18 }}
            transition={{ duration: 0.45, ease: [0.22, 0.61, 0.36, 1] }}
          >
            <Slide data={data} editable={editable} onEdit={onEdit} />
          </motion.div>
        </AnimatePresence>
      </div>

      {/* Nav */}
      <div className="w-full max-w-[1400px] mt-6 flex items-center justify-between"
        style={{ color: INK }}>
        <button
          onClick={prev}
          disabled={index === 0}
          className="px-5 py-2.5 rounded-full border bg-white hover:bg-neutral-50 disabled:opacity-40 disabled:cursor-not-allowed transition flex items-center gap-2 text-sm font-semibold"
          style={{ borderColor: HAIRLINE }}
        >
          <svg width="14" height="14" viewBox="0 0 24 24" fill="none" aria-hidden>
            <path d="M15 18 L9 12 L15 6" stroke="currentColor" strokeWidth="2.4" strokeLinecap="round" strokeLinejoin="round" />
          </svg>
          Previous
        </button>

        <div className="flex items-center gap-2">
          {SLIDES.map((_, i) => (
            <button
              key={i}
              onClick={() => setIndex(i)}
              aria-label={`Slide ${i + 1}`}
              className="transition-all"
              style={{
                width: i === index ? 32 : 8,
                height: 8,
                borderRadius: 999,
                background: i === index ? ACCENT : '#D4D4D4',
              }}
            />
          ))}
        </div>

        <button
          onClick={next}
          disabled={index === SLIDES.length - 1}
          className="px-5 py-2.5 rounded-full border bg-white hover:bg-neutral-50 disabled:opacity-40 disabled:cursor-not-allowed transition flex items-center gap-2 text-sm font-semibold"
          style={{ borderColor: HAIRLINE }}
        >
          Next
          <svg width="14" height="14" viewBox="0 0 24 24" fill="none" aria-hidden>
            <path d="M9 6 L15 12 L9 18" stroke="currentColor" strokeWidth="2.4" strokeLinecap="round" strokeLinejoin="round" />
          </svg>
        </button>
      </div>
    </div>
  );
};

### What the build delivers

- **Exactly 10 slides**, one question per slide, mounted in a single `SLIDES` array.
- **30pt-minimum equivalent typography** — body text never drops below `clamp(22px, 2vw, 30px)`; headlines hit `clamp(56px, 7vw, 120px)`.
- **Single accent** (`#0F62FE`) — every other surface is paper white, ink black, or hairline gray. No "rainbow PowerPoint" effect.
- **Custom SVG** per slide: tangled-rope pain illustration, before→after collapse, 3-node revenue flow, core+satellites magic architecture, oversized funnel, full-bleed positioning map, area + dot revenue projection, donut use-of-funds, dot timeline.

- **Framer Motion** subtle transitions (`y: 18 → 0 → -18`, 450ms cubic-bezier). Respects `prefers-reduced-motion`.
- **Keyboard nav** (← → space) and dot pagination.
- **Inline editing** on every text element via `contentEditable`; edits flow through `onEdit(path, value)`.
- **16:9 frame** (`aspect-ratio: 16/9`); export-ready for Cloudflare Browser Rendering PDF capture.
- **Investor-grade** — designed so the text is readable from the back of a conference room.
